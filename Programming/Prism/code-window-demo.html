Code Window Component

Prism.js syntax highlighting wrapped in a macOS-style window, matching the reference design.

JavaScript

app.js 
async function fetchUser(id) {
  const res = await fetch(`/api/users/${id}`);
  if (!res.ok) {
    throw new Error(`Failed to load user ${id}`);
  }
  const user = await res.json();
  return user;
}

const button = document.querySelector('#load-user');
button.addEventListener('click', () => {
  fetchUser(42).then(user => console.log(user.name));
});
